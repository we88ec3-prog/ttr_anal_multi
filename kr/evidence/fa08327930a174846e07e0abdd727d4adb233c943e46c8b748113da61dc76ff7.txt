from pathlib import Path
from build_native_index import PE

ROOT=Path(__file__).resolve().parent.parent;OUT=ROOT/'anal/native-index'
p=PE(ROOT/'Tartaros.bin');sha=hashlib.sha256(p.data).hexdigest()
assert sha=='9741d1975327b9309d82b52f631165bc410757694d7c183a5eed970a53ebef91'
md=Cs(CS_ARCH_X86,CS_MODE_32)
specs=[('actor-state-transition-dispatch',0x4ca060,0x4ca0ab),
 ('monster-state-registration-constructor',0x4d1710,0x4d18d0),

f={}
for name,s,e in specs:
 ins=list(md.disasm(p.read(s,e-s),s));assert sum(i.size for i in ins)==e-s
 assert ins[-1].mnemonic=='ret'
 starts={i.address for i in ins}
 for i in ins:
  if i.mnemonic.startswith('j') and i.op_str.startswith('0x'):
   target=int(i.op_str,16)
   if s<=target<e:assert target in starts
 f[s]={i.address:i for i in ins}
 (OUT/(name+'.txt')).write_text('\n'.join(f'{i.address:08X} {i.mnemonic:8} {i.op_str}' for i in ins)+'\n',encoding='utf-8')
checks=[(0x4ca060,0x4ca06a,'edi, 0x1a'),
 (0x4ca060,0x4ca091,'dword ptr [esi + 0x68], ecx'),
