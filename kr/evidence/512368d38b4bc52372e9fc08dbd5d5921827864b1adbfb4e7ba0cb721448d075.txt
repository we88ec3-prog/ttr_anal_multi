"""SQLite accounts and per-account saves. No plaintext credentials or tokens."""
import hashlib
import hmac
import json
import re
import secrets
import sqlite3
import time
from contextlib import contextmanager
from pathlib import Path

ITERATIONS=310000

def credentials(payload):
    if len(payload)!=112 or payload[:2]!=b'\xe8\x00':raise ValueError('Invalid login packet')
    username=payload[2:53].split(b'\0',1)[0].decode('ascii')
    digest=payload[53:104].split(b'\0',1)[0].decode('ascii').lower()
    if not re.fullmatch(r'[A-Za-z0-9_]{1,10}',username) or not re.fullmatch(r'[0-9a-f]{32}',digest):
        raise ValueError('Invalid credentials')
    return username,digest

def password_digest(password):
    if not 1<=len(password)<=50 or not password.isascii():raise ValueError('Password must be 1..50 ASCII characters')
    return hashlib.md5(password.encode('ascii')).hexdigest()

def initial_inventory():
    from new_expedition import STARTING_WEAPONS
    return {'bag':{},'equipment':{str(c):{'1':{'item_id':i,'serial':0x30000+c,'count':1}}
            for c,i in STARTING_WEAPONS.items()},'revisions':{str(c):1 for c in range(1,10)},
            'stash':{},'levels':{str(c):2 if c==4 else 1 for c in range(1,10)},'next_serial':0x40000}

class AccountStore:
    def __init__(self,path):
        self.path=Path(path);self.path.parent.mkdir(parents=True,exist_ok=True)
        with self.connect() as db:
            db.executescript('''
            PRAGMA journal_mode=WAL;
            CREATE TABLE IF NOT EXISTS schema_version(version INTEGER PRIMARY KEY);
            INSERT OR IGNORE INTO schema_version VALUES(1);
            CREATE TABLE IF NOT EXISTS accounts(id INTEGER PRIMARY KEY, username TEXT UNIQUE COLLATE NOCASE NOT NULL,
                salt BLOB NOT NULL, verifier BLOB NOT NULL, iterations INTEGER NOT NULL, created REAL NOT NULL);
            CREATE TABLE IF NOT EXISTS saves(account_id INTEGER NOT NULL REFERENCES accounts(id),
                section TEXT NOT NULL, data TEXT NOT NULL, updated REAL NOT NULL, PRIMARY KEY(account_id,section));
            CREATE TABLE IF NOT EXISTS tickets(hash BLOB PRIMARY KEY, account_id INTEGER NOT NULL REFERENCES accounts(id),
                peer TEXT NOT NULL, expires REAL NOT NULL);
            CREATE TABLE IF NOT EXISTS leases(account_id INTEGER PRIMARY KEY REFERENCES accounts(id),
                owner TEXT NOT NULL, expires REAL NOT NULL);
            CREATE TABLE IF NOT EXISTS attempts(key TEXT PRIMARY KEY, failures INTEGER NOT NULL, until REAL NOT NULL);
            ''')

    @contextmanager
    def connect(self):
        db=sqlite3.connect(self.path,timeout=10)
        db.execute('PRAGMA foreign_keys=ON')
        try:
            with db:yield db
        finally:db.close()

    def create(self,username,password,expedition=None):
        if not re.fullmatch(r'[A-Za-z0-9_]{1,10}',username):raise ValueError('Account name must be 1..10 letters/digits/underscore')
        digest=password_digest(password);salt=secrets.token_bytes(16)
        verifier=hashlib.pbkdf2_hmac('sha256',digest.encode(),salt,ITERATIONS)
        state=expedition or {'version':1,'expedition_name':'','characters':[1,3],'stage':'naming'}
        with self.connect() as db:
            ident=db.execute('INSERT INTO accounts(username,salt,verifier,iterations,created) VALUES(?,?,?,?,?)',
                             (username,salt,verifier,ITERATIONS,time.time())).lastrowid
            for section,data in [('expedition',state),('inventory',initial_inventory()),
                                 ('quests',{'accepted':False,'completed':False}),
                                 ('position',{'map':511,'character':1,'x':900.,'y':900.,'direction':64,'speed':30,'percent':100})]:
                db.execute('INSERT INTO saves VALUES(?,?,?,?)',(ident,section,json.dumps(data),time.time()))
        return ident

    def authenticate(self,username,digest,peer):
        key=peer+'|'+username.casefold();now=time.time()
        with self.connect() as db:
            attempt=db.execute('SELECT failures,until FROM attempts WHERE key=?',(key,)).fetchone()
            if attempt and attempt[0]>=5 and attempt[1]>now:return None
            row=db.execute('SELECT id,username,salt,verifier,iterations FROM accounts WHERE username=?',(username,)).fetchone()
            salt=row[2] if row else bytes(16);iterations=row[4] if row else ITERATIONS
            actual=hashlib.pbkdf2_hmac('sha256',digest.encode(),salt,iterations)
            if not row or not hmac.compare_digest(actual,row[3]):
                count=attempt[0]+1 if attempt and attempt[1]>now else 1
                db.execute('INSERT OR REPLACE INTO attempts VALUES(?,?,?)',(key,count,now+60));return None
            db.execute('DELETE FROM attempts WHERE key=?',(key,))
            # Native4C39F8 uses strcpy_s(destination,5,packet+87).
            # Four nonzero bytes plus NUL must survive both string copies.
            token=secrets.token_bytes(1)+bytes(secrets.randbelow(255)+1 for _ in range(4))+b'\0'+secrets.token_bytes(4)
            db.execute('DELETE FROM tickets WHERE expires<?',(now,))
            db.execute('INSERT INTO tickets VALUES(?,?,?,?)',(hashlib.sha256(token).digest(),row[0],peer,now+60))
            return row[0],row[1],token

    def claim(self,account_id,username,token,peer):
        with self.connect() as db:
            db.execute('BEGIN IMMEDIATE')
            row=db.execute('SELECT t.account_id FROM tickets t JOIN accounts a ON a.id=t.account_id '
                           'WHERE t.hash=? AND t.account_id=? AND substr(a.username,1,9)=? COLLATE NOCASE AND t.peer=? AND t.expires>?',
                           (hashlib.sha256(token).digest(),account_id,username,peer,time.time())).fetchone()
            if not row:return None
            db.execute('DELETE FROM tickets WHERE hash=?',(hashlib.sha256(token).digest(),))
            if db.execute('SELECT 1 FROM leases WHERE account_id=? AND expires>?',(account_id,time.time())).fetchone():return None
            owner=secrets.token_hex(16)
            db.execute('INSERT OR REPLACE INTO leases VALUES(?,?,?)',(account_id,owner,time.time()+30))
            return owner

    def heartbeat(self,ident,owner):
        with self.connect() as db:
            if db.execute('UPDATE leases SET expires=? WHERE account_id=? AND owner=?',
                          (time.time()+30,ident,owner)).rowcount!=1:raise ValueError('Account session expired')

    def release(self,ident,owner):
        with self.connect() as db:db.execute('DELETE FROM leases WHERE account_id=? AND owner=?',(ident,owner))

    def load(self,ident,section):
        with self.connect() as db:
            row=db.execute('SELECT data FROM saves WHERE account_id=? AND section=?',(ident,section)).fetchone()
            if not row:raise ValueError('Missing account save')
            return json.loads(row[0])

    def save(self,ident,section,data):
        with self.connect() as db:
            db.execute('INSERT INTO saves VALUES(?,?,?,?) ON CONFLICT(account_id,section) DO UPDATE SET data=excluded.data,updated=excluded.updated',
                       (ident,section,json.dumps(data,ensure_ascii=False),time.time()))

def login_ticket_response(result):
    import struct
    from login_protocol import login_response
    if result is None:return login_response(False)
    # Native4958D4 uses wcsncpy_s with capacity10 and count10: max9
    # characters plus NUL. Full username remains in DB; ID and ticket bind it.
    ident,name,token=result;name=name[:9];p=bytearray(login_response(True));p[2:22]=bytes(20)
    p[2:2+len(name)*2]=name.encode('utf-16le');struct.pack_into('<I',p,0x18,ident)
    p[0x86]=token[0];p[0x87:0x8c]=token[1:6];p[0x8c:0x90]=token[6:10]
    return bytes(p)

def channel_identity(payload):
    import struct
    if len(payload)!=57 or payload[:2]!=b'\x01\x00':raise ValueError('Invalid channel identity')
    raw=payload[6:26];units=struct.unpack('<10H',raw)
    name=raw[:2*(units.index(0) if 0 in units else 10)].decode('utf-16le')
    return struct.unpack_from('<I',payload,2)[0],name,payload[27:37]
