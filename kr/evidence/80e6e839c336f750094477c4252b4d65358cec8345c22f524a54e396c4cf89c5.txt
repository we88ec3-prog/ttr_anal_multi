"""Offline original battle controller sidepaths; no runtime or network use."""
import hashlib,json,struct
from pathlib import Path
from build_native_index import PE
from capstone import Cs,CS_ARCH_X86,CS_MODE_32
ROOT=Path(__file__).resolve().parent.parent;OUT=ROOT/'anal/native-index'
p=PE(ROOT/'Tartaros.bin');m=Cs(CS_ARCH_X86,CS_MODE_32)
idx=json.loads((OUT/'Tartaros.bin.json').read_text(encoding='utf-8'))
assert hashlib.sha256(p.data).hexdigest()==idx['sha256']
specs=[('single-controller-sidepath-tick',0x548a90,0x548b8e),
 ('single-controller-input-special-case',0x548a20,0x548a88),
 ('tutorial-controller-input-tick',0x548cb0,0x548d0d),
 ('actor-status-flag-query',0x4c9ed0,0x4c9f31),
 ('battle-local-target-and-motion-sidepath',0x49c5c0,0x49cc86),
 ('battle-local-target-effect-refresh',0x49cc90,0x49ce04)]
funcs={}
for name,s,e in specs:
 ins=list(m.disasm(p.read(s,e-s),s))
 assert sum(i.size for i in ins)==e-s,name
 assert ins[-1].mnemonic==('call' if s in [0x49c5c0,0x49cc90] else 'ret'),name
 bounds={i.address for i in ins}
 for i in ins:
  if i.mnemonic.startswith('j') and i.op_str.startswith('0x'):
   t=int(i.op_str,16)
   if s<=t<e:assert t in bounds,(name,hex(t))
 funcs[s]={i.address:i for i in ins}
 (OUT/(name+'.txt')).write_text('\n'.join(f'{i.address:08X} {i.mnemonic:8} {i.op_str}' for i in ins)+'\n',encoding='utf-8')
checks=[(0x548a90,0x548ab8,'dword ptr [edi + 4], 0'),
 (0x548a90,0x548ad2,'0x5452c0'),(0x548a90,0x548ad7,'esi, dword ptr [edi + 0xc]'),
 (0x548a90,0x548af2,'0x4c9ed0'),(0x548a90,0x548b19,'0x49c5c0'),
 (0x548a90,0x548b5e,'byte ptr [0xfb9011], 1'),
 (0x548a20,0x548a2a,'dword ptr [eax + 0x38c], 0x10'),
 (0x548a20,0x548a47,'0x546100'),
 (0x548cb0,0x548cbd,'dword ptr [esi + 0xdc], 0'),
 (0x548cb0,0x548cc7,'0x5452c0'),
 (0x4c9ed0,0x4c9ed1,'esi, dword ptr [ecx + 0x3f4]'),
 (0x4c9ed0,0x4c9ee0,'byte ptr [edx + 0x13], 1'),
 (0x49c5c0,0x49c74f,'eax, byte ptr [0xfb901b]'),
 (0x49c5c0,0x49cc0d,'eax, dword ptr [edx + 0x68]'),
 (0x49c5c0,0x49cc14,'0x10'),(0x49c5c0,0x49cc1c,'0x49cc90'),
 (0x49cc90,0x49ccca,'esi, edi'),(0x49cc90,0x49ccd8,'dword ptr [eax + 8], 3'),
 (0x49cc90,0x49cd44,'0x49ce10'),(0x49cc90,0x49cda7,'0x551f20')]
for s,a,op in checks:assert funcs[s][a].op_str==op,(hex(a),op)
for vt,cls,offset,target in [(0xc0fac4,'CSingleController',0x2c,0x548a90),
 (0xc0fac4,'CSingleController',0x48,0x548a20),
 (0xc0fb1c,'CTutorialController',0x2c,0x548cb0)]:
 v=next(v for v in idx['vtables'] if v['address']==vt)
 assert v['className']==f'.?AV{cls}@@' and v['entries'][offset//4]==target
network_direct=[]
for s,ins in funcs.items():
 for i in ins.values():
  if i.mnemonic=='call' and i.op_str in ['0x8069a0','0x8068b0']:
   network_direct.append(hex(i.address))
assert not network_direct
report=dict(checks='passed',sha256=idx['sha256'],boundedFunctions=len(specs),
 functions=[dict(name=n,start=hex(s),endExclusive=hex(e)) for n,s,e in specs],
 singleTick=dict(vtable='C0FAC4',slot='2C',guard='controller4 nonzero',
  normalInput='5452C0',sidepath='controllerC current actor ->state checks ->49C5C0(controller88, actor)'),
 tutorialTick=dict(vtable='C0FB1C',slot='2C',guards=['controller4 nonzero','controllerDC nonzero'],input='5452C0'),
 localMotion=dict(inputStateByte='FB901B',motion='0x10',actorVirtualSlot='68',speed=1,
  targetHelpersPending=['49D470','49CE10','49D640','49D010']),
 limitations=['No monster-specific decision producer established.',
  '49C5C0 and49CC90 include post-normal-ret throwing paths.',
  'No direct network entry call in these six functions; helpers/indirect calls not exhausted.',
  'Input-state fields and actor status enums not officially named.',
  'No packet request/reply pairing orP2P authority established.'])
(OUT/'battle-controller-sidepaths-audit.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print(json.dumps(dict(checks='passed',boundedFunctions=len(specs))))
