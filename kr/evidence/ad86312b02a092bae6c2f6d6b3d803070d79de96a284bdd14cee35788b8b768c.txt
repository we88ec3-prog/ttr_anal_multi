"""Offline KR checkpoint wire source / action notification classification."""
import hashlib
import json
from pathlib import Path
from build_native_index import PE
from capstone import Cs, CS_ARCH_X86, CS_MODE_32

ROOT=Path(__file__).resolve().parent.parent
OUT=ROOT/'anal/native-index'
p=PE(ROOT/'Tartaros.bin')
idx=json.loads((OUT/'Tartaros.bin.json').read_text(encoding='utf-8'))
assert hashlib.sha256(p.data).hexdigest()==idx['sha256']
md=Cs(CS_ARCH_X86,CS_MODE_32)
specs=[
 ('action-notification-controller-forward',0x4b19d0,0x4b19f2),
 ('action-cannot-use-skill-notification',0x545c50,0x545e19),
 ('delete-actor-list-receiver',0x4b28d0,0x4b2aed),
 ('delete-common-actor-by-id',0x4dfbc0,0x4dfc2c),
 ('checkpoint-list-receiver',0x4b2af0,0x4b2cc4),
 ('checkpoint-constructor-full',0x543700,0x543b2a),
 ('checkpoint-interaction-full',0x544070,0x544726),
 ('checkpoint-request-send',0x544020,0x544067),
 ('checkpoint-movement-update',0x4cb000,0x4cb01f),
 ('checkpoint-effect-update',0x4cb020,0x4cb04a),
]
functions={}
for name,start,end in specs:
 ins=list(md.disasm(p.read(start,end-start),start))
 assert sum(i.size for i in ins)==end-start,name
 assert ins[-1].mnemonic=='ret',name
 bounds={i.address for i in ins}
 for i in ins:
  if i.mnemonic.startswith('j') and i.op_str.startswith('0x'):
   t=int(i.op_str,16)
   if start<=t<end: assert t in bounds,(name,hex(t))
 functions[name]={i.address:i for i in ins}
 (OUT/(name+'.txt')).write_text('\n'.join(f'{i.address:08X} {i.mnemonic:8} {i.op_str}' for i in ins)+'\n',encoding='utf-8')
checks=[
 ('action-notification-controller-forward',0x4b19de,'eax, dword ptr [eax + 0x3c]'),
 ('action-cannot-use-skill-notification',0x545c7b,'eax, 0xf'),
 ('action-cannot-use-skill-notification',0x545c8f,'byte ptr [esi + 7], bl'),
 ('action-cannot-use-skill-notification',0x545c97,'0xc0f928'),

 ('checkpoint-list-receiver',0x4b2b95,'0x108'),

 ('checkpoint-list-receiver',0x4b2c9b,'edi, 0x12'),
 ('checkpoint-constructor-full',0x543738,'0x4ca730'),
 ('checkpoint-constructor-full',0x543772,'dword ptr [ebx], 0xc0f7fc'),

 ('checkpoint-interaction-full',0x5443e3,'dword ptr [ecx + eax*4 + 0xb4], 1'),
 ('checkpoint-interaction-full',0x54470e,'0x544020'),
 ('checkpoint-request-send',0x54402d,'dx, word ptr [eax + 0xf4]'),
 ('checkpoint-request-send',0x544034,'ax, word ptr [eax + 0xf6]'),
 ('checkpoint-request-send',0x54403b,'ecx, 0x1c'),

handlers={r['command']:r['handler'] for r in idx['mainPacketHandlers']}
assert handlers[0xf]==0x4b19d0 and handlers[0x1a]==0x4b28d0 and handlers[0x27]==0x4b2af0
vt=next(r for r in idx['vtables'] if r['address']==0xc0f7fc)
assert vt['className']=='.?AVCCheckPoint@@'

controllerTables=[]
for r in idx['vtables']:
 if r['className'] in ['.?AVCController@@','.?AVCFormationController@@','.?AVCSingleController@@','.?AVCTutorialController@@'] and r['thisOffset']==0:
  assert r['entries'][15]==0x545c50
  controllerTables.append(r['className'])

  'GUI callbacks/helper graphs, trigger radius and all request callers not fully expanded.',

  '001A extra header WORD purpose and000F bytes2..6 not established.',
  'No server reply pairing, monster wave trigger or P2P authority established.'])
