"""Offline spawn/control boundaries and the auxiliary minimap record path."""
from pathlib import Path
import hashlib, json, struct
from build_native_index import PE
from capstone import Cs, CS_ARCH_X86, CS_MODE_32

ROOT=Path(__file__).resolve().parent.parent
OUT=ROOT/'anal/native-index'
p=PE(ROOT/'Tartaros.bin')
idx=json.loads((OUT/'Tartaros.bin.json').read_text(encoding='utf-8'))
assert hashlib.sha256(p.data).hexdigest()==idx['sha256']
md=Cs(CS_ARCH_X86,CS_MODE_32)
def dump(start,end,name,terminal='ret'):
    ins=list(md.disasm(p.read(start,end-start),start))
    assert sum(i.size for i in ins)==end-start
    assert ins[-1].mnemonic==terminal
    bounds={i.address for i in ins}
    for i in ins:
        if i.mnemonic.startswith('j') and i.op_str.startswith('0x'):
            dest=int(i.op_str,16)
            if start<=dest<end: assert dest in bounds
    (OUT/(name+'.txt')).write_text('\n'.join(
        f'{i.address:08X} {i.mnemonic:8} {i.op_str}' for i in ins)+'\n',encoding='utf-8')
    return {i.address:i for i in ins}

# Spawn has a fatal error branch AFTER its normal ret. Jump-table bytes are data.
spawn=dump(0x4ae4d0,0x4aed8d,'monster-spawn-common-receiver','call')
register=dump(0x4dff30,0x4dffaa,'common-actor-map-registration')
owner=dump(0x4c6e50,0x4c6e9d,'player-active-actor-helper')
control=dump(0x4af920,0x4afae8,'actor-control-assignment-bounded')

render=dump(0x56f130,0x56f5b6,'spawn-auxiliary-marker-render')
# Includes the allocation-failure exception branch after the successful ret.
ctor=dump(0x56de30,0x56e0ac,'spawn-auxiliary-marker-manager-constructor','int3')
